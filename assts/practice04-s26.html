Question 4: Processes

  pid_t ret;
  int i = 1;
  printf("A %d\n", i);

  if(ret == 0) {
    i = i+1;
    printf("B %d\n", i);

    ret = fork();
    if(ret == 0) {
      i = i+2;
      printf("C %d\n", i);

    i = i+1;
