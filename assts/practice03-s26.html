Question 1: Exploits

  char text[10];
  printf("Enter a string: ");
  scanf(" %s", text);

  int len = sizeof(text);
  for (int i = 0; i < len; i++) {
    printf("%c\n", text[i]);
  }
  return 0;
}

Question 2: Conditional logic

Consider the code below.

int main() {
  int i = 10;
  while (i > 0) {
    i = i >> 1;
    printf("%d\n", i);
  }
  return 0;
}

  • What is the output of this program?
