[source]
----
#include <stdio.h>
#include <unistd.h>

int main(void) {
    int x = 1;

    printf("A: %d\n", x);

    pid_t p = fork();

    if (p == 0) {
        x += 2;
        printf("B: %d\n", x);

        x += 10;
        printf("C: %d\n", x);

        pid_t q = fork();

        if (q == 0) {
