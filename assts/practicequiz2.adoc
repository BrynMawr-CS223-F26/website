[source]
----
struct mystery {
    float x;
    float y;
};

struct node {
    int val;
    struct mystery* m;
    struct node* next;
};

struct node* insert(int val, struct node* head, struct mystery* m) {
    struct node* newNode = malloc(sizeof(struct node));
    assert(newNode != NULL); // abort program on error
    newNode->val = val;
    newNode->m = m;
    newNode->next = head;
    return newNode;
}

int main() {
    struct mystery* data = malloc(sizeof(struct mystery));
    assert(data != NULL); // abort program on error
    data->x = -2.5;
    data->y = 1.5;
    
    struct node* head = NULL;
    insert(3, head, data);
    // draw the stack here
    
    free(head);
    free(data);
    return 0;
}
